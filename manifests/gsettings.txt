# GNOME/GSettings desktop preferences. ONE source of truth for both platforms.
#
# Consumed by ./apply-gsettings.sh, called from BOTH
#   ../install.sh     (Ubuntu / macOS, run by hand)
#   ../nix/home.nix   (NixOS, as a graphical-session user unit)
#
# FORMAT
#   <schema>  <key>  <value>
#
# <value> is the REST of the line and is a GVariant literal exactly as
# `gsettings set` wants it: strings carry their own quotes ('RIGHT'), arrays
# use brackets (['<Super>1']), booleans and integers are bare.
#
# WHY NOT dconf.settings / typed Nix. home-manager's dconf module takes typed
# Nix values, which would mean translating every GVariant literal above into a
# Nix type — i.e. writing a GVariant parser in Nix, and maintaining it. Running
# the same `gsettings set` calls on both platforms needs no translation layer
# at all, so the value text below is simultaneously what Ubuntu applies and
# what NixOS applies. The cost is that this layer is imperative on NixOS too;
# it is a desktop preference, it is idempotent, and it reapplies at every
# login, so that cost is small and the alternative was a parser.
#
# TAGS
#   no-nixos    a NixOS module owns this key natively — see each note

# Alt+Tab cycles WINDOWS, one stop per window, so two Alacritty windows are two
# entries rather than one app icon to drill into. GNOME's default binds Alt+Tab
# to switch-applications and leaves switch-windows empty, so both halves have
# to move: Alt+Tab is taken off the app switcher, which keeps Super+Tab.
# Scoped to the current workspace, like the app switcher above.

org.gnome.desktop.wm.keybindings     switch-windows-backward      ['<Shift><Alt>Tab']

org.gnome.desktop.wm.keybindings     switch-applications-backward ['<Shift><Super>Tab']

# --- Power ------------------------------------------------------------------
# idle-delay / sleep-inactive-* are tagged no-nixos because the consuming host's no-sleep.nix
# owns them on this host, and there it is not a preference — the drive sits in
# an OPAL locking range that relocks on power loss, so a suspend underneath a
# running kernel is data loss. That module masks the sleep targets outright and
# sets these same GNOME keys; letting this file write them too would mean two
# writers for one safety-critical setting, and the quieter one winning by
# accident. The values happen to agree today, which is exactly how such a
# conflict stays invisible until it does not.
org.gnome.desktop.session            idle-delay                   0          no-nixos
org.gnome.settings-daemon.plugins.power sleep-inactive-ac-type    'nothing'  no-nixos
org.gnome.settings-daemon.plugins.power sleep-inactive-ac-timeout 3600       no-nixos
org.gnome.settings-daemon.plugins.power sleep-inactive-battery-timeout 7200  no-nixos
org.gnome.settings-daemon.plugins.power ambient-enabled           false

# --- Display: fractional scaling (HiDPI) ------------------------------------
# no-nixos, and NOT because something else owns the key — because on this host
# the value itself is invalid. the consuming host's desktop-env.nix records that Mutter 50.4's
# schema accepts only `kms-modifiers` and `autoclose-xwayland`:
# scale-monitor-framebuffer GRADUATED into an ordinary Settings -> Displays
# toggle and was removed from the enum. Setting it here would fail at runtime
# on GNOME 50 and is simply unnecessary.
#
# Kept (rather than deleted) because this file is shared, and the Ubuntu boxes
# still run GNOME 46/47 where the flag is both valid and required for
# fractional scaling. Revisit when those reach GNOME 50.
org.gnome.mutter                     experimental-features        ['scale-monitor-framebuffer']  no-nixos

# --- Keyboard ---------------------------------------------------------------
org.gnome.desktop.peripherals.keyboard numlock-state              false

# Snappy key repeat: 195 ms before the first repeat, 15 ms between repeats.
# These were a separate hand-rolled block in install.sh (configure_key_repeat_*)
# that read each key, compared, and set it — i.e. this script's whole job done
# a second time for two keys. The macOS half genuinely cannot live here: it
# uses `defaults write -g InitialKeyRepeat` in 15 ms ticks rather than
# GSettings in milliseconds, so install.sh keeps that part.
#
# Both values are uint32, which is why `gsettings get` prints them as
# "uint32 195" and why this script normalises type annotations before
# comparing.
org.gnome.desktop.peripherals.keyboard delay                      195
org.gnome.desktop.peripherals.keyboard repeat-interval            15
